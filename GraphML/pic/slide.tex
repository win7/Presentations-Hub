    \begin{frame}{Graphs}
        Graph is a data structure and is defined as $G = (V, E)$, where $V = \{v_1, v_2, ..., v_{n}\}$ is the nodes (or \textit{vertices}, or \textit{points}) set and $E = \{e_1, e_2, ..., e_{m}\}, e_i \in V \times V$ a set of edges (or \textit{lines}, or \textit{arcs}) formed by unordered pairs of nodes. \\

        \vspace{2mm}
        \scriptsize Number of nodes is denoted $|V|$. \\
         Number of edges is denoted $|E|$.
        
        \begin{figure}[ht!]
            \centering
            \includegraphics[width=0.25\linewidth]{pic/graph_example.pdf}
            \caption{\scriptsize Graph example.}
        \end{figure}
    \end{frame}

    \begin{frame}{Graphs}
        Furthermore, each edge $e \in E$ is said to join two vertices. If $e$ joins $u, v \in V$, then $e = (u, v)$. Vertices $u$ and $v$ are said to be adjacent. \\
        
        \begin{exampleblock}{Neighborhood}
            \begin{equation}
            N(v) = \{w \in V ∣ v \neq w, \exists \; e \in E: e = (u, v)\}
            \label{eq:neighborhood}
            \end{equation}
        \end{exampleblock}

    \begin{frame}{Breadth-First Search (BFS)}
        BFS is a graph traversal algorithm that explores nodes level by level, visiting all nodes at a given distance from the starting node before moving to the next level. \newline

        Description:
        \begin{itemize}
            \item Uses a queue (FIFO) to manage the nodes to be explored.
            \item Marks visited nodes to avoid cycles or repeated visits.
            \item Ideal for finding the shortest path in unweighted graphs.
        \end{itemize}

        \begin{note}
            {Introduce your self}
        \end{note}
    \end{frame}

    \begin{frame}{BFS algorithm}        
        \begin{columns}[t] % The "c" option specifies centered vertical alignment while the "t" option is used for top vertical alignment
            \begin{column}{0.6\textwidth}
                \begin{enumerate}
                    \item Visit the adjacent unvisited node. Mark as visited, display, and insert (queue) in a queue.
                    \item In case no adjacent node is found, remove the first node from the queue.
                    \item Repeat steps 1, and 2 until the queue is empty.
                \end{enumerate}
            \end{column}

    \begin{frame}{BFS algorithm pseudocode}   
        \begin{algorithm}[H]
            \scriptsize
            
            % \footnotesize
            \DontPrintSemicolon
            % \SetAlgoLined
            % \SetNoline
            
            \SetKwInOut{Input}{Input}
            \SetKwInOut{Output}{Output}
            
            \Input{$G=(V, E)$, $s$ \Comment*[r]{Graph $G$ and a source node $s$}}
            \Output{$D[n]$ and $Pred[n]$ \Comment*[r]{distances and predecessors of nodes}}
            % \BlankLine
            % max $\leftarrow A[0]$\;
            % $X \gets x$\;
            \For{all $v \in V - \{s\}$ \Comment*[r]{initialize all nodes except source $s$}}{ 
                $D[v] \gets \infty$\;
                $Pred[v] \gets nil$\;
            }
            $D[s] \gets 0$\;
            $Pred[s] \gets s$\;
            $Q \gets s$\;
            \While{$Q \neq \varnothing$}{
                $v \gets deque(Q)$\;
                \For{all $(u, v) \in E$ \Comment*[r]{process all neighbors of node $u$}}{
                    \If{$D[u]$ = \infty}{
                        $D[u] \gets D[v] + 1$\;
                        $Pred[u] \gets v$\;
                        $enque(Q, u)$\;
                    }
                }
            }
            % \Return max\;
            \caption{BFS.}
            \label{tab:algorithm_bfs}
        \end{algorithm}

    \begin{frame}{Depth-First Search (DFS)}
        DFS is a graph traversal algorithm that explores as deeply as possible along each branch before backtracking. \newline

        Description:
        \begin{itemize}
            \item Uses a stack (LIFO) to manage the nodes to be explored.
            \item Can be implemented recursively or iteratively.
            \item Explores a full path before returning to previous nodes.
        \end{itemize}

        \begin{note}
            {Introduce your self}
        \end{note}
    \end{frame}
    
    \begin{frame}{DFS algorithm}
        \begin{columns}[t] % The "c" option specifies centered vertical alignment while the "t" option is used for top vertical alignment
            \begin{column}{0.6\textwidth}
                \begin{enumerate}
                    \item Visit the adjacent unvisited node. Mark as visited, show, and push in a stack.
                    \item In case no adjacent nodes are found, all nodes in the stack that do not have adjacent nodes are popped.
                    \item Repeat steps 1, and 2 until the stack is empty.
                \end{enumerate}
            \end{column}

    \begin{frame}{Dijkstra's Algorithm}
        Dijkstra's algorithm finds the shortest path from a source node to all other nodes in a directed or undirected graph with non-negative edge weights. \newline

        Description:
        \begin{itemize}
            \item Uses a priority queue to select the closest unvisited node.
            \item Maintains the minimum known distance to each node.
            \item It is a weighted generalization of BFS.
        \end{itemize}

        \begin{note}
            {Introduce your self}
        \end{note}
    \end{frame}

    \begin{frame}{Dijkstra's algorithm}
        \begin{columns}[t] % The "c" option specifies centered vertical alignment while the "t" option is used for top vertical alignment
            \begin{column}{0.6\textwidth}
                \scriptsize
                Initially the source node, with its cost of $0$, is added to the unvisited set. Then the algorithm proceeds as follows as long as there is at least one node in the unvisited set.
                % 2015_Book_dataStructureAndAlgorithm
                \begin{enumerate}
                    \item Remove the node we’ll call current from the unvisited set with the least cost. All other paths to this node must have greater cost because otherwise they would have been in the unvisited set with smaller cost.
                    \item Add current to the visited set.
                    \item  For every vertex, adjacent, that is adjacent to current, check to see if adjacent is in the visited set or not. If adjacent is in the visited set, then we know the minimum cost of reaching this vertex from the source so don’t do anything.
                    \item If adjacent is not in the visited set, compute a new cost for arriving at adjacent by traversing the edge, e, from current to adjacent. A new cost can be found by adding the cost of getting to current and e‘s weight. If this new cost is better than the current cost of getting to adjacent, then update adjacent‘s cost and remember that current is the previous vertex of adjacent. Also, add adjacent to the unvisited set.
                \end{enumerate}
            \end{column}
            
            \begin{column}{0.4\textwidth}
                \begin{figure}[ht!]
                    \centering
                    \includegraphics[width=0.7\linewidth]{pic/dfs.png}

                    % \caption*{Adapted from \cite{labonne2023hands}.}
                    \label{fig:dijkstra}
                \end{figure}
            \end{column}
        \end{columns}

    \begin{comment}
        Input: Graph G(V, E) with non-negative edge weights, starting node s
        Output: Shortest path distances from s to all other nodes
        
        1. For each node v in V:
            - Set distance[v] = ∞
        2. Set distance[s] = 0
        3. Create a priority queue Q and insert (0, s)
        
        4. While Q is not empty:
            a. Pop (d, u) from Q
            b. If d > distance[u], continue
            c. For each neighbor v of u with edge weight w:
                i. If distance[v] > distance[u] + w:
                    - Set distance[v] = distance[u] + w
                    - Insert (distance[v], v) into Q
    \end{comment}

    \begin{comment}
        Dijkstra(G, start):
          for each node v in G:
            distance[v] = ∞
          distance[start] = 0
          Q = priority queue with (start, 0)
        
          while Q is not empty:
            (u, d) = Q.pop()
            for each neighbor v of u with weight w:
              if distance[v] > distance[u] + w:
                distance[v] = distance[u] + w
                Q.push((v, distance[v]))
    \end{comment}   

    \begin{frame}{Dijkstra's algorithm}        
        \begin{algorithm}[H]
            \scriptsize
    
            % \footnotesize
            \DontPrintSemicolon
            % \SetAlgoLined
            % \SetNoline
            
            \SetKwInOut{Input}{Input}
            \SetKwInOut{Output}{Output}
            
            \Input{$G$, $s$ \Comment*[r]{Graph $G$ and a source node $s$}}
            \Output{Finds shortest paths from a source node $s$ to all other nodes}
            
            % \BlankLine
            $Q \gets \varnothing$ \Comment*[r]{Create an empty priority queue}
            $D \gets \varnothing$ \Comment*[r]{Create an empty array of distances}
            
            \ForEach{$v \in G$}{
                $D[v] \gets \infty$\; % \Comment*[r]{Array of distances}
            }
            $D[s] \gets 0$\;
            $enqueue(Q, s, 0)$\;

            \While{$Q \neq \varnothing$}{
                $u, w_{min} \gets dequeue(Q)$\;
                \ForEach{$v \in N(u)$ with weight $w$} {
                    \If{$D[v] > D[u] + w$}{
                        $D[v] = D[u] + w$\;
                        $enqueue(Q, v, D[v])$\;
                    }
                }
            }
            % \Return max\;
            \caption{Dijkstra.}
            \label{tab:algorithm_dijkstra}
        \end{algorithm}
